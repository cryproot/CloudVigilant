"""
ps_shell.py — Shell interactivo via MSBuild + PowerShell Runspace
Evaluacion de seguridad autorizada. Sin lanzar powershell.exe.

Mejoras implementadas:
  - Estado persistente entre comandos (variables simples/arrays/hashtables)
  - Output en streaming (aparece linea a linea mientras ejecuta)
  - Ctrl+C cancela el comando actual sin cerrar la shell
  - Historial con flechas Arriba/Abajo (persistente entre sesiones)
  - Entrada multilínea (detecta { } ( ) [ ] sin cerrar y pipes/backtick al final)
  - Tracking de directorio actual en el prompt
  - Colores al estilo PowerShell
"""

import os, sys, re, time, ctypes, tempfile, subprocess, threading
from pathlib import Path

# ── ANSI ───────────────────────────────────────────────────────────────────────
RST    = "\033[0m"
DIM    = "\033[2m"
RED    = "\033[91m"
YELLOW = "\033[93m"
CYAN   = "\033[96m"
WHITE  = "\033[97m"
GRAY   = "\033[90m"
ANSI_RE = re.compile(r'\033\[[0-9;]*[mKJH]')

def enable_ansi():
    try:
        k32 = ctypes.windll.kernel32
        h = k32.GetStdHandle(-11)
        m = ctypes.c_ulong()
        k32.GetConsoleMode(h, ctypes.byref(m))
        k32.SetConsoleMode(h, m.value | 0x0004)
    except Exception:
        pass

def c(col, txt): return f"{col}{txt}{RST}"
def vlen(s):     return len(ANSI_RE.sub('', s))

# ── MSBuild ────────────────────────────────────────────────────────────────────
MSBUILD_PATHS = [
    r"C:\Windows\Microsoft.NET\Framework64\v4.0.30319\MSBuild.exe",
    r"C:\Windows\Microsoft.NET\Framework\v4.0.30319\MSBuild.exe",
    r"C:\Program Files\Microsoft Visual Studio\2022\Community\MSBuild\Current\Bin\MSBuild.exe",
    r"C:\Program Files (x86)\Microsoft Visual Studio\2019\BuildTools\MSBuild\Current\Bin\MSBuild.exe",
]

def find_msbuild():
    return next((p for p in MSBUILD_PATHS if os.path.exists(p)), None)

# ── Session files ──────────────────────────────────────────────────────────────
_SESS_DIR  = None   # set in main
_MSBUILD   = None
_PROC      = None
_PROC_LOCK = threading.Lock()

def _sess(name):
    return str(Path(_SESS_DIR) / name)

# ── State persistence ──────────────────────────────────────────────────────────
_PS_BUILTINS = frozenset([
    'false','true','null','_','args','ConsoleFileName','Error','Event','EventArgs',
    'EventSubscriber','ExecutionContext','foreach','HOME','Host','input',
    'LastExitCode','Matches','MyInvocation','NestedPromptLevel','PID','PROFILE',
    'PSBoundParameters','PSCmdlet','PSCommandPath','PSCulture','PSDebugContext',
    'PSDefaultParameterValues','PSEdition','PSHome','PSHOME','PSItem',
    'PSScriptRoot','PSSenderInfo','PSUICulture','PSVersionTable','PWD',
    'ShellId','StackTrace','switch','this','ConfirmPreference','DebugPreference',
    'ErrorActionPreference','ErrorView','FormatsEnumerationLimit',
    'InformationPreference','MaximumHistoryCount','OFS','OutputEncoding',
    'ProgressPreference','VerbosePreference','WarningPreference','WhatIfPreference',
    '__sf','__builtin','__s',
])

def _state_restore(sp, pp):
    return f"""
if (Test-Path '{sp}') {{
  try {{
    $__s = Get-Content '{sp}' -Raw -EA SilentlyContinue | ConvertFrom-Json
    if ($__s) {{
      $__s.PSObject.Properties | ForEach-Object {{
        try {{ Set-Variable -Name $_.Name -Value $_.Value -Scope Global -EA SilentlyContinue }} catch {{}}
      }}
    }}
  }} catch {{}}
}}
if (Test-Path '{pp}') {{
  try {{
    $__pwd = (Get-Content '{pp}' -Raw -EA SilentlyContinue).Trim().TrimStart([char]0xFEFF)
    if ($__pwd) {{ Set-Location $__pwd -EA SilentlyContinue }}
  }} catch {{}}
}}
"""

def _state_dump(sp, pp):
    blist = ','.join(f"'{b}'" for b in sorted(_PS_BUILTINS))
    return f"""
$__builtin = @({blist})
$__sf = @{{}}
Get-Variable -Scope Global -EA SilentlyContinue |
  Where-Object {{ $__builtin -notcontains $_.Name -and $_.Name -notmatch '^_' }} |
  ForEach-Object {{
    try {{
      $v = $_.Value
      if ($null -ne $v) {{
        $t = $v.GetType()
        if ($t.IsPrimitive -or $v -is [string] -or $v -is [datetime] -or
            ($v -is [array] -and $v.Length -lt 50) -or
            $v -is [System.Collections.Hashtable]) {{
          $__sf[$_.Name] = $v
        }}
      }}
    }} catch {{}}
  }}
try {{ $__sf | ConvertTo-Json -Depth 2 -Compress -EA SilentlyContinue | Set-Content '{sp}' -Encoding UTF8 -EA SilentlyContinue }} catch {{}}
try {{ (Get-Location).Path | Set-Content '{pp}' -Encoding UTF8 -EA SilentlyContinue }} catch {{}}
"""

# ── PS escaping ────────────────────────────────────────────────────────────────
def _esc(s):
    s = s.replace("\\", "\\\\")
    s = s.replace('"',  '\\"')
    s = s.replace("\r\n", "\\n")
    s = s.replace("\n",   "\\n")
    s = s.replace("\r",   "\\n")
    return s

# ── C# template ────────────────────────────────────────────────────────────────
def _build_xml(ps_esc, out_path, streaming=True, width=200):
    # streaming=True  → PowerShell class (BeginInvoke + DataAdded para output incremental)
    # streaming=False → Pipeline class  (Invoke sincrono, probado)
    # width → ancho real del terminal, pasado a Out-String -Width para formateo correcto
    if streaming:
        cs_body = f"""
              using (Runspace rs = RunspaceFactory.CreateRunspace()) {{
                rs.Open();
                using (var ps = PowerShell.Create()) {{
                  ps.Runspace = rs;
                  ps.AddScript("{ps_esc}");
                  ps.AddCommand("Out-String")
                    .AddParameter("Stream", true)
                    .AddParameter("Width", {width});
                  var output = new PSDataCollection<PSObject>();
                  output.DataAdded += (s, ev) => {{
                    var obj = ((PSDataCollection<PSObject>)s)[ev.Index];
                    if (obj != null) w.WriteLine(obj.ToString());
                  }};
                  ps.EndInvoke(ps.BeginInvoke<PSObject, PSObject>(null, output));
                  foreach (var e in ps.Streams.Error)
                    w.WriteLine("__ERR__" + e.ToString());
                }}
              }}"""
    else:
        cs_body = f"""
              using (Runspace rs = RunspaceFactory.CreateRunspace()) {{
                rs.Open();
                using (Pipeline p = rs.CreatePipeline()) {{
                  p.Commands.AddScript("{ps_esc}");
                  var osCmd = new Command("Out-String");
                  osCmd.Parameters.Add("Width", {width});
                  p.Commands.Add(osCmd);
                  foreach (var r2 in p.Invoke()) if (r2 != null) w.Write(r2.ToString());
                  if (p.Error.Count > 0)
                    foreach (var e in p.Error.ReadToEnd())
                      w.WriteLine("__ERR__" + e.ToString());
                }}
              }}"""

    return f"""<Project ToolsVersion="4.0" xmlns="http://schemas.microsoft.com/developer/msbuild/2003">
  <Target Name="Run"><T /></Target>
  <UsingTask TaskName="T" TaskFactory="CodeTaskFactory"
             AssemblyFile="$(MSBuildToolsPath)\\Microsoft.Build.Tasks.v4.0.dll">
    <Task>
      <Reference Include="System.Management.Automation" />
      <Code Type="Class" Language="cs"><![CDATA[
        using Microsoft.Build.Framework; using Microsoft.Build.Utilities;
        using System.Management.Automation; using System.Management.Automation.Runspaces;
        using System.Collections.ObjectModel; using System.Text; using System.IO;
        public class T : Task {{
          public override bool Execute() {{
            using (var w = new StreamWriter("{out_path}", false, new UTF8Encoding(false))) {{
              w.AutoFlush = true;
              try {{
                {cs_body}
              }} catch (System.Exception ex) {{
                w.WriteLine("__ERR__" + ex.Message);
              }}
            }}
            return true;
          }}
        }}
      ]]></Code>
    </Task>
  </UsingTask>
</Project>"""

# ── Streaming reader thread ────────────────────────────────────────────────────
def _stream_reader(out_file, stop_evt):
    pos = 0
    for _ in range(100):  # wait up to 5s for file creation
        if Path(out_file).exists():
            break
        time.sleep(0.05)
    else:
        return

    try:
        fh = open(out_file, 'r', encoding='utf-8', errors='replace')
        while not stop_evt.is_set():
            chunk = fh.read(8192)
            if chunk:
                pos += len(chunk)
                _write_chunk(chunk)
            else:
                time.sleep(0.05)
        # final drain
        chunk = fh.read()
        if chunk:
            _write_chunk(chunk)
        fh.close()
    except Exception:
        pass

_last_char = ['\n']  # mutable so closures can update it

def _write_chunk(text):
    text = text.replace('\r\n', '\n').replace('\r', '\n')
    parts = []
    for line in text.split('\n'):
        if line.startswith('__ERR__'):
            msg = line[7:]
            parts.append(c(RED, f'  {msg}') if msg.strip() else '')
        else:
            parts.append(c(WHITE, line))
    result = '\n'.join(parts)
    if result:
        sys.stdout.write(result)
        _last_char[0] = result[-1]
    sys.stdout.flush()

# ── Core runner ────────────────────────────────────────────────────────────────
def run_ps(script, timeout_s=60, with_state=True, stream=True):
    global _PROC
    msbuild = _MSBUILD

    tmp_out = tempfile.NamedTemporaryFile(suffix=".txt", delete=False, mode="w")
    tmp_out.close()
    op = tmp_out.name.replace("\\", "\\\\")

    if with_state:
        sp = _sess("state.json")   # paths go through _esc() once — no pre-doubling
        pp = _sess("pwd.txt")
        full_ps = _state_restore(sp, pp) + "\n" + script + "\n" + _state_dump(sp, pp)
    else:
        full_ps = script

    try:
        import shutil
        tw = shutil.get_terminal_size((200, 50)).columns
    except Exception:
        tw = 200
    xml  = _build_xml(_esc(full_ps), op, streaming=stream, width=tw)
    proj = tempfile.NamedTemporaryFile(suffix=".proj", delete=False,
                                       mode="w", encoding="utf-8")
    proj.write(xml)
    proj.close()

    stop = threading.Event()
    if stream:
        t = threading.Thread(target=_stream_reader, args=(tmp_out.name, stop), daemon=True)
        t.start()

    ok = True
    try:
        proc = subprocess.Popen(
            [msbuild, proj.name, "/nologo", "/verbosity:quiet"],
            stdout=subprocess.PIPE, stderr=subprocess.PIPE,
            shell=False, encoding="utf-8", errors="replace"
        )
        with _PROC_LOCK:
            _PROC = proc
        try:
            stdout, stderr = proc.communicate(timeout=timeout_s)
        except subprocess.TimeoutExpired:
            proc.kill()
            stdout, stderr = proc.communicate()
            ok = False
            if stream: print(c(YELLOW, '\n  [timeout]'))
        finally:
            with _PROC_LOCK:
                _PROC = None

        if stream:
            stop.set()
            t.join(timeout=2)

        if proc.returncode != 0:
            err = (stdout + stderr).strip()
            if err and not stream:
                pass  # caller handles
            elif err and stream:
                print(c(RED, f'  [rc={proc.returncode}] {err[:200]}'))
            ok = False

        if not stream:
            try:
                return ok, Path(tmp_out.name).read_text(encoding='utf-8', errors='replace')
            except Exception:
                return ok, ""

    except KeyboardInterrupt:
        with _PROC_LOCK:
            if _PROC:
                try: _PROC.terminate()
                except: pass
                _PROC = None
        if stream:
            stop.set()
            t.join(timeout=1)
        ok = False
        if stream: print(c(YELLOW, '\n  [cancelado con Ctrl+C]'))
    except Exception as e:
        ok = False
        if stream: print(c(RED, str(e)))
    finally:
        try: os.unlink(proj.name)
        except: pass
        try: os.unlink(tmp_out.name)
        except: pass

    return ok if stream else (ok, "")

def run_ps_silent(script, timeout_s=20):
    ok, out = run_ps(script, timeout_s=timeout_s, with_state=False, stream=False)
    return out.strip() if ok else ""

# ── Multiline detection ────────────────────────────────────────────────────────
def is_incomplete(text):
    depth = {'(': 0, '[': 0, '{': 0}
    close = {')': '(', ']': '[', '}': '{'}
    in_str = None
    i = 0
    while i < len(text):
        ch = text[i]
        if in_str:
            if in_str == '"' and ch == '`': i += 2; continue
            if ch == in_str: in_str = None
        elif ch in ('"', "'"):
            in_str = ch
        elif ch in depth:    depth[ch] += 1
        elif ch in close:    depth[close[ch]] = max(0, depth[close[ch]] - 1)
        i += 1
    if any(v > 0 for v in depth.values()):
        return True
    s = text.rstrip()
    return s.endswith('|') or s.endswith('`')

# ── History ────────────────────────────────────────────────────────────────────
def load_history(path, maxlen=1000):
    try:
        lines = Path(path).read_text(encoding='utf-8').splitlines()
        return [l for l in lines if l.strip()][-maxlen:]
    except Exception:
        return []

def append_history(line, path):
    try:
        with open(path, 'a', encoding='utf-8') as f:
            f.write(line + '\n')
    except Exception:
        pass

# ── Input handler (msvcrt: arrow keys, history, line editing) ─────────────────
def get_input(prompt_str, history):
    try:
        import msvcrt
    except ImportError:
        return input(prompt_str)

    pw = vlen(prompt_str)
    sys.stdout.write(prompt_str)
    sys.stdout.flush()

    buf      = []
    pos      = 0
    hidx     = len(history)
    saved    = []

    def redraw():
        try:
            import shutil
            tw = shutil.get_terminal_size((80, 24)).columns
        except Exception:
            tw = 80

        to_cursor = pw + pos          # chars visibles desde col-0 hasta el cursor
        to_end    = pw + len(buf)     # chars visibles desde col-0 hasta el final

        # Subir hasta la linea donde empieza el prompt
        lines_above = to_cursor // tw
        if lines_above > 0:
            sys.stdout.write(f'\033[{lines_above}A')
        # Col 0 + limpiar hasta el final de pantalla
        sys.stdout.write('\r\033[J')

        # Reimprimir prompt + buffer completo
        sys.stdout.write(prompt_str + ''.join(buf))

        # Reposicionar cursor si no esta al final
        if pos < len(buf):
            end_line = to_end    // tw
            cur_line = to_cursor // tw
            diff     = end_line - cur_line
            if diff > 0:
                sys.stdout.write(f'\033[{diff}A')
            col = to_cursor % tw
            sys.stdout.write('\r')
            if col > 0:
                sys.stdout.write(f'\033[{col}C')

        sys.stdout.flush()

    while True:
        try:
            ch = msvcrt.getwch()
        except Exception:
            raise KeyboardInterrupt

        if ch == '\r':                              # Enter
            print()
            return ''.join(buf)

        elif ch == '\x03':                          # Ctrl+C
            print('^C')
            raise KeyboardInterrupt

        elif ch == '\x08':                          # Backspace
            if pos > 0:
                buf.pop(pos - 1); pos -= 1; redraw()

        elif ch in ('\xe0', '\x00'):                # Special key prefix
            ch2 = msvcrt.getwch()
            if ch2 == 'H':                          # Up arrow
                if hidx == len(history): saved = buf[:]
                if hidx > 0:
                    hidx -= 1; buf = list(history[hidx]); pos = len(buf); redraw()
            elif ch2 == 'P':                        # Down arrow
                if hidx < len(history):
                    hidx += 1
                    buf  = list(history[hidx]) if hidx < len(history) else saved[:]
                    pos  = len(buf); redraw()
            elif ch2 == 'K' and pos > 0:            # Left
                pos -= 1; sys.stdout.write('\033[1D'); sys.stdout.flush()
            elif ch2 == 'M' and pos < len(buf):     # Right
                pos += 1; sys.stdout.write('\033[1C'); sys.stdout.flush()
            elif ch2 == 'G':                        # Home
                sys.stdout.write(f'\033[{pos}D') if pos else None
                pos = 0; sys.stdout.flush()
            elif ch2 == 'O':                        # End
                d = len(buf) - pos
                sys.stdout.write(f'\033[{d}C') if d else None
                pos = len(buf); sys.stdout.flush()
            elif ch2 == 'S' and pos < len(buf):     # Del key
                buf.pop(pos); redraw()

        elif ch == '\x01':                          # Ctrl+A — ir al inicio
            if pos: sys.stdout.write(f'\033[{pos}D'); pos = 0; sys.stdout.flush()
        elif ch == '\x05':                          # Ctrl+E — ir al final
            d = len(buf) - pos
            if d: sys.stdout.write(f'\033[{d}C'); pos = len(buf); sys.stdout.flush()
        elif ch == '\x0b':                          # Ctrl+K — borrar hasta el final
            del buf[pos:]; redraw()
        elif ch == '\x15':                          # Ctrl+U — borrar hasta el inicio
            del buf[:pos]; pos = 0; redraw()
        elif ch == '\x0c':                          # Ctrl+L — limpiar pantalla
            sys.stdout.write("\033[2J\033[H")
            sys.stdout.write(prompt_str + ''.join(buf))
            d = len(buf) - pos
            if d: sys.stdout.write(f'\033[{d}D')
            sys.stdout.flush()
        elif ord(ch) >= 32:                         # Carácter normal
            buf.insert(pos, ch); pos += 1
            if pos == len(buf):
                sys.stdout.write(ch); sys.stdout.flush()
            else:
                redraw()

# ── Prompt ─────────────────────────────────────────────────────────────────────
def read_pwd():
    try:
        p = Path(_sess("pwd.txt"))
        if p.exists():
            # utf-8-sig strips BOM que PS 5.x añade con Set-Content -Encoding UTF8
            return p.read_text(encoding='utf-8-sig').strip()
    except Exception:
        pass
    return os.environ.get("USERPROFILE", "C:\\")

def make_prompt(pwd):
    return f"{c(CYAN,'PS')} {c(WHITE, pwd)}{c(CYAN,'>')} "

# ── Banner ─────────────────────────────────────────────────────────────────────
BANNER = f"""
{c(WHITE, 'Windows PowerShell')}
{c(GRAY,  'Copyright (C) Microsoft Corporation. Todos los derechos reservados.')}

{c(DIM+GRAY, 'Engine : MSBuild.exe + C# Runspace  (sin powershell.exe)')}
{c(DIM+GRAY, 'Estado : persiste entre comandos  |  Streaming: activo')}
{c(DIM+GRAY, 'Teclas : ↑↓ historial  Ctrl+C cancela  Ctrl+L limpia  { multilinea')}
"""

# ── Main ───────────────────────────────────────────────────────────────────────
def main():
    global _SESS_DIR, _MSBUILD

    if sys.platform != "win32":
        print("ABORTAR: requiere Windows"); return

    enable_ansi()

    _MSBUILD = find_msbuild()
    if not _MSBUILD:
        print(c(RED, "ERROR: MSBuild.exe no encontrado")); input(); return

    # Directorio de sesion en %TEMP%
    _SESS_DIR = os.path.join(tempfile.gettempdir(), "pssess")
    os.makedirs(_SESS_DIR, exist_ok=True)

    hist_path = _sess("history.txt")
    history   = load_history(hist_path)

    print(BANNER)

    # Inicializar pwd.txt con el directorio actual de Python (no el de MSBuild/TEMP)
    init_pwd = os.getcwd()
    try:
        Path(_sess("pwd.txt")).write_text(init_pwd, encoding='utf-8')
    except Exception:
        pass
    current_pwd = init_pwd

    while True:
        # Leer la linea (con soporte multilinea)
        prompt = make_prompt(current_pwd)
        try:
            line = get_input(prompt, history)
        except KeyboardInterrupt:
            print()
            continue

        cmd = line.strip()
        if not cmd:
            continue

        # Comandos internos
        cmd_l = cmd.lower()
        if cmd_l in ("exit", "quit", "q"):
            break
        if cmd_l in ("cls", "clear"):
            sys.stdout.write("\033[2J\033[H"); sys.stdout.flush(); continue
        if cmd_l == "help":
            print(c(CYAN, "\n  Atajos de teclado:"))
            for k, v in [("↑ / ↓", "navegar historial"),
                         ("← / →", "mover cursor"),
                         ("Inicio / Fin", "ir al principio / final"),
                         ("Ctrl+A / E", "inicio / final de línea"),
                         ("Ctrl+K", "borrar hasta el final"),
                         ("Ctrl+U", "borrar hasta el inicio"),
                         ("Ctrl+L", "limpiar pantalla"),
                         ("Ctrl+C", "cancelar comando actual"),
                         ("{ sin cerrar", "modo multilínea — Intro para seguir")]:
                print(f"    {c(YELLOW, k):30s} {c(GRAY, v)}")
            print()
            continue

        # Acumular bloques multilinea
        full_cmd = cmd
        while is_incomplete(full_cmd):
            cont_prompt = f"{c(GRAY, '>>')} "
            try:
                more = get_input(cont_prompt, history)
            except KeyboardInterrupt:
                full_cmd = ""; break
            full_cmd += "\n" + more

        if not full_cmd.strip():
            continue

        # Guardar en historial
        history.append(full_cmd)
        append_history(full_cmd, hist_path)

        # Ejecutar
        _last_char[0] = '\n'
        t0 = time.time()
        run_ps(full_cmd, timeout_s=120, with_state=True, stream=True)
        elapsed = time.time() - t0

        # Asegurar que el output termina en su propia linea
        if _last_char[0] != '\n':
            sys.stdout.write('\n')
            sys.stdout.flush()

        # Actualizar prompt con nuevo pwd (sin BOM)
        new_pwd = read_pwd()
        if new_pwd:
            current_pwd = new_pwd

        if elapsed > 3.0:
            print(c(GRAY, f"  [{elapsed:.1f}s]"))

    print(c(GRAY, "\nSesion terminada.\n"))

if __name__ == "__main__":
    main()
